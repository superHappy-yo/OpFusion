# 按功能目录收集自有实现；具体算子仍由同名头源自动发现，不维护名称名单。
set(contractsSources "")
foreach(group IN ITEMS common graph registry platform model plan operators)
  file(GLOB groupSources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/${group}/*.cpp")
  list(APPEND contractsSources ${groupSources})
endforeach()
list(SORT contractsSources)
add_library(OpFusionContracts STATIC ${contractsSources})
target_include_directories(OpFusionContracts PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(OpFusionContracts PUBLIC cxx_std_17)
target_compile_options(OpFusionContracts PRIVATE -Wall -Wextra -Wpedantic)

# 生成物仅写入构建目录；内容摘要跟随源码、路径及构建条件变化。
set(registrationSource "${CMAKE_CURRENT_BINARY_DIR}/BuiltinOperators.cpp")
opfusionBuiltinOperators(OpFusionContracts "${registrationSource}")
target_sources(OpFusionContracts PRIVATE "${registrationSource}")
